\begin{tikzpicture}[font=\small,
  doc/.style={draw=black!55, thick, fill=blue!7, rounded corners=3pt,
              minimum width=3.2cm, minimum height=0.85cm, align=center,
              font=\scriptsize},
  val/.style={draw=black!40, fill=gray!10, rounded corners=2pt,
              minimum width=2.2cm, minimum height=0.7cm,
              align=center, font=\scriptsize},
  txt/.style={draw=orange!75!black, fill=orange!15, rounded corners=2pt,
              minimum width=2.2cm, minimum height=0.7cm,
              align=center, font=\scriptsize},
  arr/.style={-{Stealth[length=5pt,width=4pt]}, thick, draw=black!55},
  tarr/.style={-{Stealth[length=5pt,width=4pt]}, thick, draw=orange!75!black}
]
\node[doc] (root) at (6.0, 0) {%
  \texttt{\{"name": "Alice",}\\
  \texttt{"score": 42,}\\
  \texttt{"address": \{...\}\}}};

\node[val] (name)  at (0.0,  -2.5) {\texttt{"Alice"}};
\node[val] (score) at (3.5,  -2.5) {\texttt{42}};
\node[doc] (addr)  at (7.5,  -2.5) {\texttt{\{"city": "Paris"\}}};
\node[val] (city)  at (12.0, -2.5) {\texttt{"Paris"}};

\node[txt] (nametxt) at (0.0,  -4.5) {\texttt{"Alice"}\\{\tiny text result}};
\node[txt] (citytxt) at (12.0, -4.5) {\texttt{"Paris"}\\{\tiny text result}};

\draw[arr] (root) -- node[above, sloped, font=\scriptsize] {\texttt{-> 'name'}}    (name);
\draw[arr] (root) -- node[above, sloped, font=\scriptsize] {\texttt{-> 'score'}}   (score);
\draw[arr] (root) -- node[above, sloped, font=\scriptsize] {\texttt{-> 'address'}} (addr);
\draw[arr] (addr.east) -- node[above, font=\scriptsize]    {\texttt{-> 'city'}}    (city.west);

\draw[tarr] (root) .. controls (-3, -1) and (-3, -4) ..
  node[left, font=\scriptsize, pos=0.5] {\texttt{->> 'name'}} (nametxt);
\draw[tarr] (addr.south) .. controls (7.5, -5.5) and (9.5, -4.5) ..
  node[above, font=\scriptsize, pos=0.5] {\texttt{->> 'city'}} (citytxt.west);
\end{tikzpicture}
\caption{JSONB navigation operators.}
\end{figure}
